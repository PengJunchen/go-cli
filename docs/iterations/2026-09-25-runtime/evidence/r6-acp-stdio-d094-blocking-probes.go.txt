package acp

import (
	"bytes"
	"context"
	"io"
	"sync"
	"testing"
	"time"

	"github.com/stretchr/testify/require"
)

func TestReviewStdioReaderPanicJoinsBeforeDisconnectDone(t *testing.T) {
	if !runStdioPanicChild(t, "review-blocking-reader") {
		return
	}
	reader := &reviewBlockingPanicReader{started: make(chan struct{}), release: make(chan struct{})}
	client := NewStdioAdapter(reader, &bytes.Buffer{})
	require.NoError(t, client.Connect(context.Background()))
	select {
	case <-reader.started:
	case <-time.After(time.Second):
		t.Fatal("reader did not enter Read")
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	err := client.Disconnect(ctx)
	pendingDone := requireStdioPending(t, err, false)
	assertChannelOpen(t, pendingDone, "reader panic callback remains in-flight")
	close(reader.release)
	assertChannelCloses(t, pendingDone)
	require.ErrorIs(t, client.SendMessage(context.Background(), ACPMessage{Type: TypeMessage, Content: "must-not-write"}), errStdioNotConnected)
	require.EqualError(t, client.Disconnect(context.Background()), "acp: stdio reader panicked")
}

func TestReviewStdioWriterPanicJoinsBeforeWriteDone(t *testing.T) {
	if !runStdioPanicChild(t, "review-blocking-writer") {
		return
	}
	peerReader, peerWriter := io.Pipe()
	writer := &reviewBlockingPanicWriter{started: make(chan struct{}), release: make(chan struct{})}
	client := NewStdioAdapter(peerReader, writer)
	t.Cleanup(func() {
		closeIgnored(peerWriter)
		closeIgnored(peerReader)
	})
	require.NoError(t, client.Connect(context.Background()))
	ctx, cancel := context.WithCancel(context.Background())
	sendResult := make(chan error, 1)
	go func() { sendResult <- client.SendMessage(ctx, ACPMessage{Type: TypeMessage, Content: "private"}) }()
	select {
	case <-writer.started:
	case <-time.After(time.Second):
		t.Fatal("writer did not enter Write")
	}
	cancel()
	pendingDone := requireStdioPending(t, <-sendResult, true)
	assertChannelOpen(t, pendingDone, "writer panic callback remains in-flight")
	close(writer.release)
	assertChannelCloses(t, pendingDone)
	closeIgnored(peerWriter)
	require.EqualError(t, client.Disconnect(context.Background()), "acp: stdio writer panicked after dispatch; delivery unknown")
}

func TestReviewStdioCloserPanicDoesNotClaimReaderResourceClosed(t *testing.T) {
	if !runStdioPanicChild(t, "review-blocking-closer") {
		return
	}
	peerReader, peerWriter := io.Pipe()
	readerStarted := make(chan struct{})
	reader := &signalReadStarted{reader: peerReader, started: readerStarted}
	closer := &reviewBlockingPanicCloser{started: make(chan struct{}), release: make(chan struct{}), unwound: make(chan struct{})}
	client := NewStdioAdapter(reader, &bytes.Buffer{}, WithStdioOwnedTransportCloser(closer))
	t.Cleanup(func() {
		closeIgnored(peerWriter)
		closeIgnored(peerReader)
	})
	require.NoError(t, client.Connect(context.Background()))
	select {
	case <-readerStarted:
	case <-time.After(time.Second):
		t.Fatal("reader did not enter Read")
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	pendingDone := requireStdioPending(t, client.Disconnect(ctx), false)
	select {
	case <-closer.started:
	case <-time.After(time.Second):
		t.Fatal("closer did not enter Close")
	}
	assertChannelOpen(t, pendingDone, "closer panic callback remains in-flight")
	close(closer.release)
	select {
	case <-closer.unwound:
	case <-time.After(time.Second):
		t.Fatal("closer callback did not unwind")
	}
	assertChannelOpen(t, pendingDone, "a recovered Close panic is not proof that the blocked reader resource closed")
	closeIgnored(peerWriter)
	assertChannelCloses(t, pendingDone)
	require.EqualError(t, client.Disconnect(context.Background()), "acp: stdio transport closer panicked")
}

type reviewBlockingPanicReader struct {
	started chan struct{}
	release chan struct{}
	once    sync.Once
}

func (r *reviewBlockingPanicReader) Read([]byte) (int, error) {
	r.once.Do(func() { close(r.started) })
	<-r.release
	panic("review-reader-private")
}

type reviewBlockingPanicWriter struct {
	mu      sync.Mutex
	calls   int
	started chan struct{}
	release chan struct{}
}

func (w *reviewBlockingPanicWriter) Write(p []byte) (int, error) {
	w.mu.Lock()
	w.calls++
	calls := w.calls
	w.mu.Unlock()
	if calls == 1 {
		return len(p), nil
	}
	close(w.started)
	<-w.release
	panic("review-writer-private")
}

type reviewBlockingPanicCloser struct {
	started chan struct{}
	release chan struct{}
	unwound chan struct{}
}

func (c *reviewBlockingPanicCloser) Close() error {
	close(c.started)
	defer close(c.unwound)
	<-c.release
	panic("review-closer-private")
}

var _ io.Closer = (*reviewBlockingPanicCloser)(nil)
