package main

import (
 "archive/zip"
 "bytes"
 "encoding/xml"
 "fmt"
 "io"
 "os"
 "strings"
)

const oldStatus = "Release status: INITIAL"
const releaseStatus = "Release status: INITIAL" // target: live_review_3A61DCA80B0C.pptx

var titles = []string{"LIVE_REVIEW_3A61DCA80B0C", "What Was Checked", "Tool Cycle and Retained Facts", "Release Status"}
var facts = []string{"Synthetic checks completed: 66 of 100", "Synthetic median tool cycle: 2.3 seconds", "Synthetic review facts retained: 7 of 10"}
const disclosure = "Synthetic QA inputs only - not product measurements"

func must(err error) { if err != nil { panic(err) } }
func esc(s string) string { var b bytes.Buffer; must(xml.EscapeText(&b, []byte(s))); return b.String() }
func shape(id int, text string, y, size int) string {
 return fmt.Sprintf(`<p:sp><p:nvSpPr><p:cNvPr id="%d" name="Text %d"/><p:cNvSpPr txBox="1"/><p:nvPr/></p:nvSpPr><p:spPr><a:xfrm><a:off x="685800" y="%d"/><a:ext cx="10820400" cy="914400"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom><a:noFill/><a:ln><a:noFill/></a:ln></p:spPr><p:txBody><a:bodyPr wrap="square"/><a:lstStyle/><a:p><a:r><a:rPr lang="en-US" sz="%d"><a:solidFill><a:srgbClr val="18354A"/></a:solidFill><a:latin typeface="Arial"/></a:rPr><a:t>%s</a:t></a:r><a:endParaRPr lang="en-US"/></a:p></p:txBody></p:sp>`, id,id,y,size,esc(text))
}
func slide(title string, lines []string) string {
 s := `<?xml version="1.0" encoding="UTF-8" standalone="yes"?><p:sld xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"><p:cSld><p:bg><p:bgPr><a:solidFill><a:srgbClr val="F5F8FA"/></a:solidFill><a:effectLst/></p:bgPr></p:bg><p:spTree><p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr>`
 s += shape(2,title,457200,3200)
 for i,line := range lines { s += shape(i+3,line,1828800+i*1143000,2400) }
 s += shape(len(lines)+3,disclosure,5486400,1600)
 return s+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sld>`
}
func create(path string) {
 f,err := os.Create(path); must(err)
 z := zip.NewWriter(f)
 put := func(name,body string) { w,e := z.Create(name); must(e); _,e = io.WriteString(w,body); must(e) }
 ct := `<?xml version="1.0"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/><Override PartName="/ppt/presentation.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.presentation.main+xml"/><Override PartName="/ppt/slideMasters/slideMaster1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideMaster+xml"/><Override PartName="/ppt/slideLayouts/slideLayout1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideLayout+xml"/>`
 for i:=1;i<=4;i++ { ct += fmt.Sprintf(`<Override PartName="/ppt/slides/slide%d.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slide+xml"/>`,i) }
 put("[Content_Types].xml",ct+`</Types>`)
 rels := func(body string) string { return `<?xml version="1.0"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">`+body+`</Relationships>` }
 rel := func(id,kind,target string) string { return fmt.Sprintf(`<Relationship Id="%s" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/%s" Target="%s"/>`,id,kind,target) }
 put("_rels/.rels",rels(rel("rId1","officeDocument","ppt/presentation.xml")))
 p := `<?xml version="1.0"?><p:presentation xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><p:sldMasterIdLst><p:sldMasterId id="2147483648" r:id="rId5"/></p:sldMasterIdLst><p:sldIdLst>`
 pr := ""
 for i:=1;i<=4;i++ { p += fmt.Sprintf(`<p:sldId id="%d" r:id="rId%d"/>`,255+i,i); pr += rel(fmt.Sprintf("rId%d",i),"slide",fmt.Sprintf("slides/slide%d.xml",i)) }
 put("ppt/presentation.xml",p+`</p:sldIdLst><p:sldSz cx="12192000" cy="6858000"/><p:notesSz cx="6858000" cy="9144000"/></p:presentation>`)
 put("ppt/_rels/presentation.xml.rels",rels(pr+rel("rId5","slideMaster","slideMasters/slideMaster1.xml")))
 tree := `<p:spTree><p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr></p:spTree>`
 ns := ` xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main" xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"`
 put("ppt/slideMasters/slideMaster1.xml",`<?xml version="1.0"?><p:sldMaster`+ns+`><p:cSld>`+tree+`</p:cSld><p:clrMap accent1="accent1" accent2="accent2" accent3="accent3" accent4="accent4" accent5="accent5" accent6="accent6" bg1="lt1" bg2="lt2" folHlink="folHlink" hlink="hlink" tx1="dk1" tx2="dk2"/><p:sldLayoutIdLst><p:sldLayoutId id="2147483649" r:id="rId1"/></p:sldLayoutIdLst><p:txStyles/></p:sldMaster>`)
 put("ppt/slideMasters/_rels/slideMaster1.xml.rels",rels(rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml")))
 put("ppt/slideLayouts/slideLayout1.xml",`<?xml version="1.0"?><p:sldLayout`+ns+` type="blank" preserve="1"><p:cSld name="Blank">`+tree+`</p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sldLayout>`)
 put("ppt/slideLayouts/_rels/slideLayout1.xml.rels",rels(rel("rId1","slideMaster","../slideMasters/slideMaster1.xml")))
 bodies := [][]string{{"Synthetic QA briefing"},{facts[0]},{facts[1],facts[2]},{releaseStatus}}
 for i:=1;i<=4;i++ { put(fmt.Sprintf("ppt/slides/slide%d.xml",i),slide(titles[i-1],bodies[i-1])); put(fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels",i),rels(rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml"))) }
 must(z.Close()); must(f.Close())
}
func update(path string) {
 z,err := zip.OpenReader(path); must(err)
 var out bytes.Buffer
 w := zip.NewWriter(&out)
 found := false
 for _,f := range z.File {
  if f.Name != "ppt/slides/slide4.xml" { must(w.Copy(f)); continue }
  r,e := f.Open(); must(e); b,e := io.ReadAll(r); must(e); must(r.Close())
  if !bytes.Contains(b,[]byte(esc(oldStatus))) { panic("slide 4 old status missing") }
  b = bytes.Replace(b,[]byte(esc(oldStatus)),[]byte(esc(releaseStatus)),1)
  h := f.FileHeader; dst,e := w.CreateHeader(&h); must(e); _,e = dst.Write(b); must(e); found = true
 }
 if !found { panic("slide 4 missing") }
 must(w.Close()); must(z.Close())
 must(os.WriteFile(path,out.Bytes(),0644))
}
func verify(path string) {
 z,e := zip.OpenReader(path); must(e); defer z.Close()
 texts := make(map[string][]string)
 count := 0
 for _,f := range z.File {
  r,e := f.Open(); must(e); b,e := io.ReadAll(r); must(e); must(r.Close()) // Reading validates ZIP CRC.
  if !strings.HasSuffix(f.Name,".xml") && !strings.HasSuffix(f.Name,".rels") { continue }
  d := xml.NewDecoder(bytes.NewReader(b))
  for { tok,e := d.Token(); if e == io.EOF { break }; must(e)
   if start,ok := tok.(xml.StartElement); ok && start.Name.Local=="t" && start.Name.Space=="http://schemas.openxmlformats.org/drawingml/2006/main" { var text string; must(d.DecodeElement(&text,&start)); texts[f.Name]=append(texts[f.Name],text) }
   if start,ok := tok.(xml.StartElement); ok && f.Name=="ppt/presentation.xml" && start.Name.Local=="sldId" { count++ }
  }
 }
 if count!=4 { panic(fmt.Sprintf("unexpected slide count: %d",count)) }
 all := ""
 for i,title := range titles {
  name := fmt.Sprintf("ppt/slides/slide%d.xml",i+1)
  t := texts[name]
  if len(t)==0 || t[0]!=title { panic("incorrect title: "+name) }
  joined := strings.Join(t,"\n")
  if !strings.Contains(joined,disclosure) { panic("missing disclosure") }
  all += joined+"\n"
 }
 for _,fact := range facts { if !strings.Contains(all,fact) { panic("missing metric: "+fact) } }
 if !strings.Contains(strings.Join(texts["ppt/slides/slide4.xml"],"\n"),releaseStatus) { panic("missing release status") }
 fmt.Printf("Verified saved ZIP: %s\nSlide count: %d\n",path,count)
 for i:=1;i<=4;i++ { fmt.Printf("Slide %d editable text:\n%s\n",i,strings.Join(texts[fmt.Sprintf("ppt/slides/slide%d.xml",i)],"\n")) }
}
func main() {
 if len(os.Args)!=2 { panic("usage: create_live_deck.go target.pptx") }
 path := os.Args[1]
 _,e := os.Stat(path)
 if os.IsNotExist(e) { create(path) } else { must(e); update(path) }
 verify(path)
}
