package core

import (
	"context"
	"testing"

	"github.com/stretchr/testify/require"
	"github.com/pengjunchen/go-cli/internal/mock"
)

// TestAudit_ContinuationStreamErrorIsReportedAsSuccess captures the current
// behavior: the initial response is truncated, the continuation fails, and
// LoopAgent.Run returns the partial answer with nil error.
func TestAudit_ContinuationStreamErrorIsReportedAsSuccess(t *testing.T) {
	model := mock.NewMockLLMServer(mock.NewConversationTemplate(
		"AUDIT-CONTINUATION", "continuation error",
		mock.ConversationTurn{AssistantContent: "partial answer", FinishReason: "length"},
		mock.ConversationTurn{AssistantError: "continuation network failure"},
	))
	loop := NewLoopAgent(WithLLM(model))

	events, err := loop.Run(context.Background(), Submission{Content: "answer"})
	require.NoError(t, err) // currently unexpected; should become explicit partial/failure status
	require.Len(t, findEvents(events, "message"), 1)
	require.Equal(t, "partial answer", findEvents(events, "message")[0])
	require.Equal(t, 2, model.CallCount())
}
