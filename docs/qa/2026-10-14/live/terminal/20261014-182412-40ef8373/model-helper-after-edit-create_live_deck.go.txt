package main

import (
 "archive/zip"
 "bytes"
 "encoding/xml"
 "fmt"
 "io"
 "os"
 "strings"
)

const oldStatus = "Release status: INITIAL"
const releaseStatus = "Release status: REVISED" // target: live_review_E6A0DAEABA5F.pptx
const disclosure = "Synthetic QA inputs only - not product measurements"
var titles = []string{"LIVE_REVIEW_E6A0DAEABA5F", "What Was Checked", "Tool Cycle and Retained Facts", "Release Status"}
var facts = []string{"Synthetic checks completed: 50 of 100", "Synthetic median tool cycle: 3.2 seconds", "Synthetic review facts retained: 7 of 10"}
const head = `<?xml version="1.0" encoding="UTF-8" standalone="yes"?>`
const relNS = "http://schemas.openxmlformats.org/package/2006/relationships"
const base = "http://schemas.openxmlformats.org/officeDocument/2006/relationships/"
const ns = `xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"`
const group = `<p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr>`
func esc(s string) string { var b bytes.Buffer; xml.EscapeText(&b, []byte(s)); return b.String() }
func rel(id, kind, target string) string { return fmt.Sprintf(`<Relationship Id="%s" Type="%s%s" Target="%s"/>`,id,base,kind,target) }
func rels(s string) string { return head+`<Relationships xmlns="`+relNS+`">`+s+`</Relationships>` }
func shape(id int, text string, y, h, size int, color string, bold bool) string {
 b := "0"; if bold { b = "1" }
 return fmt.Sprintf(`<p:sp><p:nvSpPr><p:cNvPr id="%d" name="Text %d"/><p:cNvSpPr txBox="1"/><p:nvPr/></p:nvSpPr><p:spPr><a:xfrm><a:off x="650000" y="%d"/><a:ext cx="10900000" cy="%d"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom><a:noFill/><a:ln><a:noFill/></a:ln></p:spPr><p:txBody><a:bodyPr wrap="square"/><a:lstStyle/><a:p><a:r><a:rPr lang="en-US" sz="%d" b="%s"><a:solidFill><a:srgbClr val="%s"/></a:solidFill><a:latin typeface="Arial"/></a:rPr><a:t>%s</a:t></a:r><a:endParaRPr lang="en-US"/></a:p></p:txBody></p:sp>`,id,id,y,h,size,b,color,esc(text))
}
func create(path string) error {
 entries := map[string]string{}
 ct := head+`<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/>`
 for _, v := range []struct{p,t string}{{"presentation","presentation"},{"slideMasters/slideMaster1","slideMaster"},{"slideLayouts/slideLayout1","slideLayout"}} { ct += `<Override PartName="/ppt/`+v.p+`.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.`+v.t+`+xml"/>` }
 ct += `<Override PartName="/ppt/theme/theme1.xml" ContentType="application/vnd.openxmlformats-officedocument.theme+xml"/>`
 ids, pr := "", rel("rId1","slideMaster","slideMasters/slideMaster1.xml")
 bodies := [][]string{{"Synthetic QA briefing",disclosure},{facts[0],disclosure},{facts[1],facts[2],disclosure},{releaseStatus,disclosure}}
 for i,title := range titles {
 n := i+1
 ct += fmt.Sprintf(`<Override PartName="/ppt/slides/slide%d.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slide+xml"/>`,n)
 ids += fmt.Sprintf(`<p:sldId id="%d" r:id="rId%d"/>`,255+n,n+1)
 pr += rel(fmt.Sprintf("rId%d",n+1),"slide",fmt.Sprintf("slides/slide%d.xml",n))
 content := shape(2,title,500000,1100000,3200,"17324D",true)
 for j,t := range bodies[i] { y,size,color := 2100000+j*1000000,2400,"243746"; if t == disclosure { y,size,color = 5700000,1600,"526575" }; content += shape(j+3,t,y,800000,size,color,false) }
 entries[fmt.Sprintf("ppt/slides/slide%d.xml",n)] = head+`<p:sld `+ns+`><p:cSld><p:bg><p:bgPr><a:solidFill><a:srgbClr val="F7F9FC"/></a:solidFill><a:effectLst/></p:bgPr></p:bg><p:spTree>`+group+content+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sld>`
 entries[fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels",n)] = rels(rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml"))
 }
 entries["[Content_Types].xml"] = ct+`</Types>`
 entries["_rels/.rels"] = rels(rel("rId1","officeDocument","ppt/presentation.xml"))
 entries["ppt/presentation.xml"] = head+`<p:presentation `+ns+`><p:sldMasterIdLst><p:sldMasterId id="2147483648" r:id="rId1"/></p:sldMasterIdLst><p:sldIdLst>`+ids+`</p:sldIdLst><p:sldSz cx="12192000" cy="6858000" type="screen16x9"/><p:notesSz cx="6858000" cy="9144000"/></p:presentation>`
 entries["ppt/_rels/presentation.xml.rels"] = rels(pr)
 entries["ppt/slideMasters/slideMaster1.xml"] = head+`<p:sldMaster `+ns+`><p:cSld><p:spTree>`+group+`</p:spTree></p:cSld><p:clrMap accent1="accent1" accent2="accent2" accent3="accent3" accent4="accent4" accent5="accent5" accent6="accent6" bg1="lt1" bg2="lt2" folHlink="folHlink" hlink="hlink" tx1="dk1" tx2="dk2"/><p:sldLayoutIdLst><p:sldLayoutId id="2147483649" r:id="rId1"/></p:sldLayoutIdLst><p:txStyles><p:titleStyle/><p:bodyStyle/><p:otherStyle/></p:txStyles></p:sldMaster>`
 entries["ppt/slideMasters/_rels/slideMaster1.xml.rels"] = rels(rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml")+rel("rId2","theme","../theme/theme1.xml"))
 entries["ppt/slideLayouts/slideLayout1.xml"] = head+`<p:sldLayout `+ns+` type="blank" preserve="1"><p:cSld name="Blank"><p:spTree>`+group+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sldLayout>`
 entries["ppt/slideLayouts/_rels/slideLayout1.xml.rels"] = rels(rel("rId1","slideMaster","../slideMasters/slideMaster1.xml"))
 colors := ""; for i,k := range []string{"dk1","lt1","dk2","lt2","accent1","accent2","accent3","accent4","accent5","accent6","hlink","folHlink"} { colors += `<a:`+k+`><a:srgbClr val="`+[]string{"17324D","FFFFFF","243746","F7F9FC","246B99","408060","806040","604080","408080","808040","0000FF","800080"}[i]+`"/></a:`+k+`>` }
 fill := `<a:solidFill><a:schemeClr val="phClr"/></a:solidFill>`
 entries["ppt/theme/theme1.xml"] = head+`<a:theme xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" name="Plain"><a:themeElements><a:clrScheme name="Plain">`+colors+`</a:clrScheme><a:fontScheme name="Arial"><a:majorFont><a:latin typeface="Arial"/><a:ea typeface=""/><a:cs typeface=""/></a:majorFont><a:minorFont><a:latin typeface="Arial"/><a:ea typeface=""/><a:cs typeface=""/></a:minorFont></a:fontScheme><a:fmtScheme name="Plain"><a:fillStyleLst>`+strings.Repeat(fill,3)+`</a:fillStyleLst><a:lnStyleLst>`+strings.Repeat(`<a:ln w="12700">`+fill+`<a:prstDash val="solid"/></a:ln>`,3)+`</a:lnStyleLst><a:effectStyleLst>`+strings.Repeat(`<a:effectStyle><a:effectLst/></a:effectStyle>`,3)+`</a:effectStyleLst><a:bgFillStyleLst>`+strings.Repeat(fill,3)+`</a:bgFillStyleLst></a:fmtScheme></a:themeElements></a:theme>`
 f,err := os.Create(path); if err != nil { return err }; z := zip.NewWriter(f)
 for name,data := range entries { w,e := z.Create(name); if e != nil { f.Close(); return e }; if _,e = io.WriteString(w,data); e != nil { f.Close(); return e } }
 if err = z.Close(); err != nil { f.Close(); return err }; return f.Close()
}
func update(path string) error {
 z,err := zip.OpenReader(path); if err != nil { return err }; defer z.Close()
 var out bytes.Buffer; w := zip.NewWriter(&out); found := false
 for _,f := range z.File {
 if f.Name != "ppt/slides/slide4.xml" { if err = w.Copy(f); err != nil { return err }; continue }
 r,e := f.Open(); if e != nil { return e }; b,e := io.ReadAll(r); r.Close(); if e != nil { return e }
 old,new := "<a:t>"+esc(oldStatus)+"</a:t>","<a:t>"+esc(releaseStatus)+"</a:t>"
 if bytes.Count(b,[]byte(old)) != 1 { return fmt.Errorf("expected exactly one slide-4 status") }; found = true
 if old == new { if err = w.Copy(f); err != nil { return err }; continue }
 header := f.FileHeader; dst,e := w.CreateHeader(&header); if e != nil { return e }; if _,e = dst.Write(bytes.Replace(b,[]byte(old),[]byte(new),1)); e != nil { return e }
 }
 if !found { return fmt.Errorf("slide 4 missing") }; if err = w.Close(); err != nil { return err }
 // An unchanged status leaves the entire original package untouched.
 if oldStatus == releaseStatus { return nil }; return os.WriteFile(path,out.Bytes(),0644)
}
func verify(path string) error {
 z,err := zip.OpenReader(path); if err != nil { return err }; defer z.Close()
 slides := map[string][]string{}; all := ""; count := 0
 for _,f := range z.File {
 r,e := f.Open(); if e != nil { return e }; b,e := io.ReadAll(r); r.Close(); if e != nil { return e }
 if !strings.HasSuffix(f.Name,".xml") && !strings.HasSuffix(f.Name,".rels") { continue }
 d := xml.NewDecoder(bytes.NewReader(b)); texts := []string{}
 for { tok,e := d.Token(); if e == io.EOF { break }; if e != nil { return fmt.Errorf("%s: %w",f.Name,e) }; if s,ok := tok.(xml.StartElement); ok && s.Name.Local == "t" && s.Name.Space == "http://schemas.openxmlformats.org/drawingml/2006/main" { var t string; if e = d.DecodeElement(&t,&s); e != nil { return e }; texts = append(texts,t) } }
 if strings.HasPrefix(f.Name,"ppt/slides/slide") && strings.HasSuffix(f.Name,".xml") { count++; slides[f.Name] = texts; all += strings.Join(texts,"\n")+"\n"; if !bytes.Contains(b,[]byte("<p:sp>")) { return fmt.Errorf("editable shapes missing") } }
 }
 if count != 4 { return fmt.Errorf("slide count: %d",count) }
 for i,t := range titles { ts := slides[fmt.Sprintf("ppt/slides/slide%d.xml",i+1)]; if len(ts)==0 || ts[0]!=t { return fmt.Errorf("slide %d title mismatch",i+1) } }
 for _,t := range append(append([]string{},facts...),disclosure) { if !strings.Contains(all,t+"\n") { return fmt.Errorf("missing required text: %s",t) } }
 if !strings.Contains(strings.Join(slides["ppt/slides/slide4.xml"],"\n"),releaseStatus) { return fmt.Errorf("status missing") }
 fmt.Printf("Verified saved file: %s\nSlide count: %d\nAll four exact titles, synthetic metric lines, disclosure, slide-4 status, and editable text shapes verified. ZIP entries read and XML parsed successfully.\n",path,count)
 return nil
}
func main() {
 if len(os.Args)!=2 { fmt.Fprintln(os.Stderr,"usage: create_live_deck target.pptx"); os.Exit(1) }
 path := os.Args[1]; var err error
 if _,e := os.Stat(path); os.IsNotExist(e) { err = create(path) } else if e != nil { err = e } else { err = update(path) }
 if err == nil { err = verify(path) }; if err != nil { fmt.Fprintln(os.Stderr,err); os.Exit(1) }
}
