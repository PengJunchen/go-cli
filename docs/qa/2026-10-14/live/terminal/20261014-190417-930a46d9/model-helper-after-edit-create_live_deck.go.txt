package main

import (
 "archive/zip"
 "bytes"
 "encoding/xml"
 "fmt"

 "os"
 "strings"
)

const oldStatus = "Release status: INITIAL"
const releaseStatus = "Release status: REVISED" // target: live_review_A18BD3BD0108.pptx
const mime = "application/vnd.openxmlformats-officedocument.presentationml.presentation.main+xml"
var titles = []string{"LIVE_REVIEW_A18BD3BD0108", "What Was Checked", "Tool Cycle and Retained Facts", "Release Status"}
func esc(s string) string { var b bytes.Buffer; if err := xml.EscapeText(&b, []byte(s)); err != nil { panic(err) }; return b.String() }
func shape(id int, text string, y, size int) string {
 return fmt.Sprintf(`<p:sp><p:nvSpPr><p:cNvPr id="%d" name="Text %d"/><p:cNvSpPr txBox="1"/><p:nvPr/></p:nvSpPr><p:spPr><a:xfrm><a:off x="640000" y="%d"/><a:ext cx="10900000" cy="850000"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom><a:noFill/></p:spPr><p:txBody><a:bodyPr wrap="square"/><a:lstStyle/><a:p><a:r><a:rPr lang="en-US" sz="%d"><a:solidFill><a:srgbClr val="18324B"/></a:solidFill><a:latin typeface="Arial"/></a:rPr><a:t>%s</a:t></a:r><a:endParaRPr lang="en-US"/></a:p></p:txBody></p:sp>`, id,id,y,size,esc(text))
}
func slide(title string, lines []string) string {
 s := `<p:sld xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"><p:cSld><p:bg><p:bgPr><a:solidFill><a:srgbClr val="F5F8FB"/></a:solidFill><a:effectLst/></p:bgPr></p:bg><p:spTree><p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr>`
 s += shape(2,title,550000,3200)
 for i,line := range lines { s += shape(i+3,line,1900000+i*1050000,2200) }
 return s+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sld>`
}
func create(path string) error {
 entries := map[string]string{}
 ct := `<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/><Override PartName="/ppt/presentation.xml" ContentType="`+mime+`"/><Override PartName="/ppt/slideMasters/slideMaster1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideMaster+xml"/><Override PartName="/ppt/slideLayouts/slideLayout1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideLayout+xml"/>`
 pres := `<p:presentation xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"><p:sldMasterIdLst><p:sldMasterId id="2147483648" r:id="rId5"/></p:sldMasterIdLst><p:sldIdLst>`
 rel := `<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">`
 bodies := [][]string{{"Synthetic QA briefing", "Synthetic QA inputs only - not product measurements"},{"Synthetic checks completed: 54 of 100", "Synthetic QA inputs only - not product measurements"},{"Synthetic median tool cycle: 4.3 seconds", "Synthetic review facts retained: 8 of 10", "Synthetic QA inputs only - not product measurements"},{releaseStatus,"Synthetic QA inputs only - not product measurements"}}
 for i,t := range titles {
 n := i+1
 ct += fmt.Sprintf(`<Override PartName="/ppt/slides/slide%d.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slide+xml"/>`,n)
 pres += fmt.Sprintf(`<p:sldId id="%d" r:id="rId%d"/>`,255+n,n)
 rel += fmt.Sprintf(`<Relationship Id="rId%d" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/slide" Target="slides/slide%d.xml"/>`,n,n)
 entries[fmt.Sprintf("ppt/slides/slide%d.xml",n)] = slide(t,bodies[i])
 entries[fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels",n)] = `<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/slideLayout" Target="../slideLayouts/slideLayout1.xml"/></Relationships>`
 }
 entries["[Content_Types].xml"] = ct+`</Types>`
 entries["_rels/.rels"] = `<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="ppt/presentation.xml"/></Relationships>`
 entries["ppt/presentation.xml"] = pres+`</p:sldIdLst><p:sldSz cx="12192000" cy="6858000"/><p:notesSz cx="6858000" cy="9144000"/></p:presentation>`
 entries["ppt/_rels/presentation.xml.rels"] = rel+`<Relationship Id="rId5" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/slideMaster" Target="slideMasters/slideMaster1.xml"/></Relationships>`
 group := `<p:spTree><p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr></p:spTree>`
 ns := ` xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"`
 entries["ppt/slideMasters/slideMaster1.xml"] = `<p:sldMaster`+ns+`><p:cSld>`+group+`</p:cSld><p:clrMap accent1="accent1" accent2="accent2" accent3="accent3" accent4="accent4" accent5="accent5" accent6="accent6" bg1="lt1" bg2="lt2" folHlink="folHlink" hlink="hlink" tx1="dk1" tx2="dk2"/><p:sldLayoutIdLst><p:sldLayoutId id="2147483649" r:id="rId1"/></p:sldLayoutIdLst><p:txStyles><p:titleStyle/><p:bodyStyle/><p:otherStyle/></p:txStyles></p:sldMaster>`
 entries["ppt/slideMasters/_rels/slideMaster1.xml.rels"] = `<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/slideLayout" Target="../slideLayouts/slideLayout1.xml"/></Relationships>`
 entries["ppt/slideLayouts/slideLayout1.xml"] = `<p:sldLayout`+ns+` type="blank" preserve="1"><p:cSld name="Blank">`+group+`</p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sldLayout>`
 entries["ppt/slideLayouts/_rels/slideLayout1.xml.rels"] = `<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/slideMaster" Target="../slideMasters/slideMaster1.xml"/></Relationships>`
 f,err := os.Create(path); if err != nil { return err }; z := zip.NewWriter(f)
 for name,data := range entries { w,e := z.Create(name); if e != nil { return e }; if _,e = io.WriteString(w,data); e != nil { return e } }
 if err = z.Close(); err != nil { f.Close(); return err }; return f.Close()
}
func update(path string) error {
 r,err := zip.OpenReader(path); if err != nil { return err }; defer r.Close()
 var out bytes.Buffer; z := zip.NewWriter(&out); found := false
 for _,f := range r.File {
 if f.Name != "ppt/slides/slide4.xml" || oldStatus == releaseStatus { if err = z.Copy(f); err != nil { return err }; if f.Name == "ppt/slides/slide4.xml" { found = true }; continue }
 rc,e := f.Open(); if e != nil { return e }; data,e := io.ReadAll(rc); rc.Close(); if e != nil { return e }
 if bytes.Count(data,[]byte(esc(oldStatus))) != 1 { return fmt.Errorf("expected one slide-4 status") }
 data = bytes.Replace(data,[]byte(esc(oldStatus)),[]byte(esc(releaseStatus)),1)
 h := f.FileHeader; w,e := z.CreateHeader(&h); if e != nil { return e }; if _,e = w.Write(data); e != nil { return e }; found = true
 }
 if !found { return fmt.Errorf("slide 4 missing") }; if err = z.Close(); err != nil { return err }
 return os.WriteFile(path,out.Bytes(),0644)
}
func verify(path string) error {
 r,err := zip.OpenReader(path); if err != nil { return err }; defer r.Close()
 texts := map[string][]string{}; count := 0; mimeOK := false; ids := 0
 for _,f := range r.File {
 rc,e := f.Open(); if e != nil { return e }; data,e := io.ReadAll(rc); rc.Close(); if e != nil { return e }
 if !strings.HasSuffix(f.Name,".xml") && !strings.HasSuffix(f.Name,".rels") { continue }
 d := xml.NewDecoder(bytes.NewReader(data))
 for { tok,e := d.Token(); if e == io.EOF { break }; if e != nil { return fmt.Errorf("%s: %w",f.Name,e) }; start,ok := tok.(xml.StartElement); if !ok { continue }
 if f.Name == "[Content_Types].xml" && start.Name.Local == "Override" { part,typ := "",""; for _,a := range start.Attr { if a.Name.Local == "PartName" { part=a.Value }; if a.Name.Local == "ContentType" { typ=a.Value } }; if part=="/ppt/presentation.xml" { mimeOK=typ==mime } }
 if f.Name == "ppt/presentation.xml" && start.Name.Local == "sldId" { ids++ }
 if start.Name.Local == "t" && start.Name.Space == "http://schemas.openxmlformats.org/drawingml/2006/main" { var text string; if e=d.DecodeElement(&text,&start); e!=nil { return e }; texts[f.Name]=append(texts[f.Name],text) }
 }
 if strings.HasPrefix(f.Name,"ppt/slides/slide") && strings.HasSuffix(f.Name,".xml") { count++ }
 }
 if !mimeOK || count!=4 || ids!=4 { return fmt.Errorf("invalid presentation: MIME=%v slides=%d references=%d",mimeOK,count,ids) }
 all := ""
 for i,t := range titles { name:=fmt.Sprintf("ppt/slides/slide%d.xml",i+1); lines:=texts[name]; if len(lines)==0 || lines[0]!=t { return fmt.Errorf("incorrect title on slide %d",i+1) }; all+=strings.Join(lines,"\n")+"\n" }
 for _,s := range []string{"Synthetic checks completed: 54 of 100","Synthetic median tool cycle: 4.3 seconds","Synthetic review facts retained: 8 of 10","Synthetic QA inputs only - not product measurements"} { if !strings.Contains(all,s+"\n") { return fmt.Errorf("missing text: %s",s) } }
 if !strings.Contains(strings.Join(texts["ppt/slides/slide4.xml"],"\n"),releaseStatus) { return fmt.Errorf("missing release status") }
 fmt.Printf("Verified saved file: %s\nSlide count: %d\nPresentation MIME: %s\nAll required titles, synthetic facts, disclosure, and slide-4 status verified in editable text shapes.\n",path,count,mime)
 return nil
}
func main() {
 if len(os.Args)!=2 { fmt.Fprintln(os.Stderr,"usage: create_live_deck target.pptx"); os.Exit(1) }
 path:=os.Args[1]; _,err:=os.Stat(path)
 if os.IsNotExist(err) { err=create(path) } else if err==nil { err=update(path) }
 if err==nil { err=verify(path) }; if err!=nil { fmt.Fprintln(os.Stderr,err); os.Exit(1) }
}
