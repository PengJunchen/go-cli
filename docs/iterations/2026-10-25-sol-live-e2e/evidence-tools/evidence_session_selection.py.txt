"""Select the exact physical session observed in the owned UI."""
import re

def select_session_files(files, observed_sid=None):
    files=list(files)
    if observed_sid is not None and re.fullmatch(r's_[0-9a-f]{32}',observed_sid) is None:
        raise ValueError('invalid observed physical session ID')
    if any(re.fullmatch(r's_[0-9a-f]{32}',path.stem) is None for path in files):
        raise ValueError('unexpected durable session filename')
    ids={path.stem for path in files}
    if observed_sid is None:
        if len(ids)!=1:
            raise ValueError('an explicit UI-observed physical SID is required')
        observed_sid=next(iter(ids))
    selected=[path for path in files if path.stem==observed_sid]
    if not selected:
        raise ValueError('observed physical SID has no durable file')
    return selected

def appended_rows(data, baseline):
    """Read only an appended JSONL suffix after verifying the exact old prefix."""
    import hashlib
    import json
    offset=baseline['bytes'] if baseline else 0
    if not isinstance(offset,int) or offset<0 or offset>len(data):
        raise ValueError('invalid or truncated evidence prefix')
    if baseline and hashlib.sha256(data[:offset]).hexdigest()!=baseline['sha256']:
        raise ValueError('evidence prefix was rewritten')
    if offset and data[offset-1:offset]!=b'\n':
        raise ValueError('evidence baseline does not end at a JSONL line boundary')
    suffix=data[offset:]
    if suffix and not suffix.endswith(b'\n'):
        raise ValueError('evidence suffix is incomplete')
    try:
        return [json.loads(line) for line in suffix.splitlines() if line]
    except (ValueError,UnicodeError):
        raise ValueError('owned evidence suffix could not be parsed') from None
