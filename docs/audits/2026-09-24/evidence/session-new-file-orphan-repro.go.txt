package main

import (
	"context"
	"fmt"
	"os"
	"time"

	"github.com/pengjunchen/go-cli/internal/session"
)

func main() {
	ctx := context.Background()
	dir, err := os.MkdirTemp("", "go-cli-session-repro-")
	must(err)
	defer os.RemoveAll(dir)

	store := session.NewJSONLSessionStore(dir)
	must(store.Open(ctx))
	must(store.SetSessionID("session-a", true))
	must(store.Append(ctx, &session.SessionEntry{ID: "entry-a", Type: session.EntryTypeUser, Content: "A", Timestamp: time.Now()}))
	must(store.Save(ctx))

	tree, err := session.NewDefaultSessionTreeBuilder().BuildFromStore(ctx, store)
	must(err)
	oldLeaf := tree.CurrentLeaf()

	must(store.SetSessionID("session-b", true))
	must(store.Append(ctx, &session.SessionEntry{ID: "entry-b", ParentID: oldLeaf, Type: session.EntryTypeUser, Content: "B", Timestamp: time.Now()}))
	must(store.Save(ctx))
	must(store.Close())

	reopened := session.NewJSONLSessionStore(dir)
	must(reopened.Open(ctx))
	defer reopened.Close()
	entries, err := reopened.List(ctx)
	must(err)
	recoveredTree, err := session.NewDefaultSessionTreeBuilder().BuildFromStore(ctx, reopened)
	must(err)
	fmt.Printf("old_leaf=%q new_file_entries=%d recovered_leaf=%q\n", oldLeaf, len(entries), recoveredTree.CurrentLeaf())
}

func must(err error) {
	if err != nil {
		panic(err)
	}
}
