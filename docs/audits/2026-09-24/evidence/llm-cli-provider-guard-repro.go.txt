package cli

import (
	"context"
	"testing"

	"github.com/pengjunchen/go-cli/internal/llm"
	"github.com/pengjunchen/go-cli/internal/production"
)

func TestAuditProviderResolutionForClaude(t *testing.T) {
	if _, _, _, err := buildModel(context.Background(), nil, "claude", "claude-test"); err != nil {
		t.Fatalf("expected configured native Claude provider to resolve, got: %v", err)
	}
}

type auditReproModel struct{}

func (auditReproModel) Generate(context.Context, []llm.Message, ...llm.Option) (*llm.Message, error) {
	return &llm.Message{Role: llm.RoleAssistant, Content: "sensitive-output"}, nil
}

func (auditReproModel) Stream(context.Context, []llm.Message, ...llm.Option) (<-chan llm.MessageChunk, error) {
	ch := make(chan llm.MessageChunk, 1)
	ch <- llm.MessageChunk{Role: llm.RoleAssistant, Content: "sensitive-output", Final: true}
	close(ch)
	return ch, nil
}

type auditReproGuard struct{}

func (auditReproGuard) Check(context.Context, string) (*production.GuardResult, error) {
	return &production.GuardResult{Allowed: false, Sanitized: "[blocked]"}, nil
}

func (auditReproGuard) Name() string { return "audit-repro" }

func TestAuditOutputGuardProtectsStream(t *testing.T) {
	ch, err := (&outputGuardModel{inner: auditReproModel{}, guard: auditReproGuard{}}).Stream(context.Background(), nil)
	if err != nil {
		t.Fatal(err)
	}
	for chunk := range ch {
		if chunk.Content == "sensitive-output" {
			t.Fatal("unguarded model content escaped through Stream")
		}
	}
}
