package main

import (
 "archive/zip"
 "bytes"
 "encoding/xml"
 "fmt"
 "io"
 "os"
 "strings"
)

const oldStatus = "Release status: INITIAL"
const releaseStatus = "Release status: INITIAL" // target: live_review_E08CCC4BB0E4.pptx
const mime = "application/vnd.openxmlformats-officedocument.presentationml.presentation.main+xml"
const ns = `xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"`
var titles = []string{"LIVE_REVIEW_E08CCC4BB0E4", "What Was Checked", "Tool Cycle and Retained Facts", "Release Status"}
var facts = []string{"Synthetic checks completed: 59 of 100", "Synthetic median tool cycle: 3.4 seconds", "Synthetic review facts retained: 7 of 10"}
const disclosure = "Synthetic QA inputs only - not product measurements"
func must(err error) { if err != nil { panic(err) } }
func esc(s string) string { var b bytes.Buffer; must(xml.EscapeText(&b, []byte(s))); return b.String() }
func rels(body string) string { return `<?xml version="1.0" encoding="UTF-8"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">`+body+`</Relationships>` }
func rel(id, kind, target string) string { return `<Relationship Id="`+id+`" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/`+kind+`" Target="`+target+`"/>` }
const group = `<p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr>`
func shape(id int, text string, y, height, size int, bold bool) string {
 b := ""; if bold { b = ` b="1"` }
 return fmt.Sprintf(`<p:sp><p:nvSpPr><p:cNvPr id="%d" name="Text %d"/><p:cNvSpPr txBox="1"/><p:nvPr/></p:nvSpPr><p:spPr><a:xfrm><a:off x="650000" y="%d"/><a:ext cx="10900000" cy="%d"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom><a:noFill/><a:ln><a:noFill/></a:ln></p:spPr><p:txBody><a:bodyPr wrap="square"/><a:lstStyle/><a:p><a:r><a:rPr lang="en-US" sz="%d"%s><a:solidFill><a:srgbClr val="17324D"/></a:solidFill><a:latin typeface="Arial"/></a:rPr><a:t>%s</a:t></a:r><a:endParaRPr lang="en-US"/></a:p></p:txBody></p:sp>`,id,id,y,height,size,b,esc(text))
}
func slide(title string, lines []string) string {
 s := `<?xml version="1.0" encoding="UTF-8"?><p:sld `+ns+`><p:cSld><p:bg><p:bgPr><a:solidFill><a:srgbClr val="F5F7FA"/></a:solidFill><a:effectLst/></p:bgPr></p:bg><p:spTree>`+group+shape(2,title,550000,1200000,3200,true)
 for i,line := range lines { s += shape(i+3,line,2050000+i*1000000,800000,2400,false) }
 s += shape(len(lines)+3,disclosure,5950000,500000,1500,false)
 return s+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sld>`
}
func create(path string) {
 entries := map[string]string{}
 ct := `<?xml version="1.0" encoding="UTF-8"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/><Override PartName="/ppt/presentation.xml" ContentType="`+mime+`"/><Override PartName="/ppt/slideMasters/slideMaster1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideMaster+xml"/><Override PartName="/ppt/slideLayouts/slideLayout1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideLayout+xml"/>`
 ids := ""
 pr := rel("rId1","slideMaster","slideMasters/slideMaster1.xml")
 bodies := [][]string{{"Synthetic QA review"},{facts[0]},{facts[1],facts[2]},{releaseStatus}}
 for i := 1; i <= 4; i++ {
  ct += fmt.Sprintf(`<Override PartName="/ppt/slides/slide%d.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slide+xml"/>`,i)
  ids += fmt.Sprintf(`<p:sldId id="%d" r:id="rId%d"/>`,255+i,i+1)
  pr += rel(fmt.Sprintf("rId%d",i+1),"slide",fmt.Sprintf("slides/slide%d.xml",i))
  entries[fmt.Sprintf("ppt/slides/slide%d.xml",i)] = slide(titles[i-1],bodies[i-1])
  entries[fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels",i)] = rels(rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml"))
 }
 entries["[Content_Types].xml"] = ct+`</Types>`
 entries["_rels/.rels"] = rels(rel("rId1","officeDocument","ppt/presentation.xml"))
 entries["ppt/presentation.xml"] = `<?xml version="1.0" encoding="UTF-8"?><p:presentation `+ns+`><p:sldMasterIdLst><p:sldMasterId id="2147483648" r:id="rId1"/></p:sldMasterIdLst><p:sldIdLst>`+ids+`</p:sldIdLst><p:sldSz cx="12192000" cy="6858000" type="screen16x9"/><p:notesSz cx="6858000" cy="9144000"/></p:presentation>`
 entries["ppt/_rels/presentation.xml.rels"] = rels(pr)
 entries["ppt/slideMasters/slideMaster1.xml"] = `<?xml version="1.0" encoding="UTF-8"?><p:sldMaster `+ns+`><p:cSld><p:spTree>`+group+`</p:spTree></p:cSld><p:clrMap accent1="accent1" accent2="accent2" accent3="accent3" accent4="accent4" accent5="accent5" accent6="accent6" bg1="lt1" bg2="lt2" folHlink="folHlink" hlink="hlink" tx1="dk1" tx2="dk2"/><p:sldLayoutIdLst><p:sldLayoutId id="2147483649" r:id="rId1"/></p:sldLayoutIdLst><p:txStyles><p:titleStyle/><p:bodyStyle/><p:otherStyle/></p:txStyles></p:sldMaster>`
 entries["ppt/slideMasters/_rels/slideMaster1.xml.rels"] = rels(rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml"))
 entries["ppt/slideLayouts/slideLayout1.xml"] = `<?xml version="1.0" encoding="UTF-8"?><p:sldLayout `+ns+` type="blank" preserve="1"><p:cSld name="Blank"><p:spTree>`+group+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sldLayout>`
 entries["ppt/slideLayouts/_rels/slideLayout1.xml.rels"] = rels(rel("rId1","slideMaster","../slideMasters/slideMaster1.xml"))
 f,err := os.Create(path); must(err); w := zip.NewWriter(f)
 for name,data := range entries { out,err := w.Create(name); must(err); _,err = io.WriteString(out,data); must(err) }
 must(w.Close()); must(f.Close())
}
func update(path string) {
 r,err := zip.OpenReader(path); must(err)
 var buf bytes.Buffer; w := zip.NewWriter(&buf); found := false
 for _,f := range r.File {
  if f.Name != "ppt/slides/slide4.xml" { must(w.Copy(f)); continue }
  rd,err := f.Open(); must(err); data,err := io.ReadAll(rd); must(err); must(rd.Close())
  if !bytes.Contains(data,[]byte(esc(oldStatus))) { panic("slide 4 status missing") }
  found = true
  if oldStatus == releaseStatus { must(w.Copy(f)); continue }
  data = bytes.Replace(data,[]byte(esc(oldStatus)),[]byte(esc(releaseStatus)),1)
  out,err := w.CreateHeader(&f.FileHeader); must(err); _,err = out.Write(data); must(err)
 }
 must(w.Close()); must(r.Close()); if !found { panic("slide 4 missing") }
 if oldStatus != releaseStatus { must(os.WriteFile(path,buf.Bytes(),0644)) }
}
func verify(path string) {
 r,err := zip.OpenReader(path); must(err); defer r.Close()
 entries := map[string][]byte{}
 for _,f := range r.File {
  rd,err := f.Open(); must(err); data,err := io.ReadAll(rd); must(err); must(rd.Close()); entries[f.Name] = data
  if strings.HasSuffix(f.Name,".xml") || strings.HasSuffix(f.Name,".rels") {
   d := xml.NewDecoder(bytes.NewReader(data)); for { _,e := d.Token(); if e == io.EOF { break }; must(e) }
  }
 }
 var ct struct { Overrides []struct { Part string `xml:"PartName,attr"`; Type string `xml:"ContentType,attr"` } `xml:"Override"` }
 must(xml.Unmarshal(entries["[Content_Types].xml"],&ct)); good := false
 for _,o := range ct.Overrides { if o.Part == "/ppt/presentation.xml" && o.Type == mime { good = true } }; if !good { panic("presentation MIME mismatch") }
 d := xml.NewDecoder(bytes.NewReader(entries["ppt/presentation.xml"])); count := 0
 for { t,e := d.Token(); if e == io.EOF { break }; must(e); if s,ok := t.(xml.StartElement); ok && s.Name.Local == "sldId" { count++ } }
 if count != 4 { panic("expected four slides") }
 all := ""
 for i,title := range titles {
  data,ok := entries[fmt.Sprintf("ppt/slides/slide%d.xml",i+1)]; if !ok { panic("missing slide") }
  dec := xml.NewDecoder(bytes.NewReader(data)); texts := []string{}; shapes := 0
  for { tok,e := dec.Token(); if e == io.EOF { break }; must(e); if s,ok := tok.(xml.StartElement); ok { if s.Name.Local == "sp" { shapes++ }; if s.Name.Local == "t" { var text string; must(dec.DecodeElement(&text,&s)); texts = append(texts,text) } } }
  if shapes < 3 || len(texts) < 3 || texts[0] != title { panic("editable title or shapes missing") }
  joined := strings.Join(texts,"\n"); if !strings.Contains(joined,disclosure) { panic("disclosure missing") }
  if i == 3 && !strings.Contains(joined,releaseStatus) { panic("status missing") }; all += joined+"\n"
 }
 for _,fact := range facts { if !strings.Contains(all,fact) { panic("fixture fact missing") } }
 abs,err := os.Getwd(); must(err)
 fmt.Printf("Verified saved PPTX: %s/%s\nSlide count: %d\nEditable text shapes and all required fixture text verified.\nPresentation MIME: %s\n",abs,path,count,mime)
}
func main() {
 if len(os.Args) != 2 { panic("usage: create_live_deck.go target.pptx") }; path := os.Args[1]
 _,err := os.Stat(path); if os.IsNotExist(err) { create(path) } else { must(err); update(path) }; verify(path)
}
