package main

import (
 "archive/zip"
 "bytes"
 "encoding/xml"
 "fmt"

 "os"

const oldStatus = "Release status: INITIAL"
const releaseStatus = "Release status: REVISED" // target: live_review_DF0447F6240F.pptx
const mime = "application/vnd.openxmlformats-officedocument.presentationml.presentation.main+xml"
var titles = []string{"LIVE_REVIEW_DF0447F6240F", "What Was Checked", "Tool Cycle and Retained Facts", "Release Status"}
const disclosure = "Synthetic QA inputs only - not product measurements"
var facts = []string{"Synthetic checks completed: 68 of 100", "Synthetic median tool cycle: 4.4 seconds", "Synthetic review facts retained: 10 of 10"}
const head = `<?xml version="1.0" encoding="UTF-8" standalone="yes"?>`

func must(err error) { if err != nil { panic(err) } }
func esc(s string) string { var b bytes.Buffer; must(xml.EscapeText(&b, []byte(s))); return b.String() }
func shape(id int, text string, y, size int, color string) string {
 return fmt.Sprintf(`<p:sp><p:nvSpPr><p:cNvPr id="%d" name="Text %d"/><p:cNvSpPr txBox="1"/><p:nvPr/></p:nvSpPr><p:spPr><a:xfrm><a:off x="685800" y="%d"/><a:ext cx="10820400" cy="914400"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom><a:noFill/></p:spPr><p:txBody><a:bodyPr wrap="square"/><a:lstStyle/><a:p><a:r><a:rPr lang="en-US" sz="%d"><a:solidFill><a:srgbClr val="%s"/></a:solidFill><a:latin typeface="Arial"/></a:rPr><a:t>%s</a:t></a:r><a:endParaRPr lang="en-US"/></a:p></p:txBody></p:sp>`, id,id,y,size,color,esc(text))
}
func slide(title string, lines []string) string {
 s := head+`<p:sld xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"><p:cSld><p:bg><p:bgPr><a:solidFill><a:srgbClr val="F5F7FA"/></a:solidFill><a:effectLst/></p:bgPr></p:bg><p:spTree><p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr>`
 s += shape(2,title,548640,3000,"16324F")
 for i,line := range lines { s += shape(i+3,line,1828800+i*1005840,2400,"243746") }
 s += shape(len(lines)+3,disclosure,5669280,1400,"526575")
 return s+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sld>`
}
const relHead = `<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">`
func rel(id, kind, target string) string { return fmt.Sprintf(`<Relationship Id="%s" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/%s" Target="%s"/>`,id,kind,target) }
func create(path string) {
 entries := map[string]string{}
 ct := head+`<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/><Override PartName="/ppt/presentation.xml" ContentType="`+mime+`"/>`
 for i:=1;i<=4;i++ { ct+=fmt.Sprintf(`<Override PartName="/ppt/slides/slide%d.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slide+xml"/>`,i) }
 ct+=`<Override PartName="/ppt/slideMasters/slideMaster1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideMaster+xml"/><Override PartName="/ppt/slideLayouts/slideLayout1.xml" ContentType="application/vnd.openxmlformats-officedocument.presentationml.slideLayout+xml"/></Types>`
 entries["[Content_Types].xml"]=ct
 entries["_rels/.rels"]=head+relHead+rel("rId1","officeDocument","ppt/presentation.xml")+`</Relationships>`
 p:=head+`<p:presentation xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><p:sldMasterIdLst><p:sldMasterId id="2147483648" r:id="rId5"/></p:sldMasterIdLst><p:sldIdLst>`
 pr:=head+relHead
 for i:=1;i<=4;i++ { p+=fmt.Sprintf(`<p:sldId id="%d" r:id="rId%d"/>`,255+i,i); pr+=rel(fmt.Sprintf("rId%d",i),"slide",fmt.Sprintf("slides/slide%d.xml",i)) }
 entries["ppt/presentation.xml"]=p+`</p:sldIdLst><p:sldSz cx="12192000" cy="6858000"/><p:notesSz cx="6858000" cy="9144000"/></p:presentation>`
 entries["ppt/_rels/presentation.xml.rels"]=pr+rel("rId5","slideMaster","slideMasters/slideMaster1.xml")+`</Relationships>`
 group:=`<p:nvGrpSpPr><p:cNvPr id="1" name=""/><p:cNvGrpSpPr/><p:nvPr/></p:nvGrpSpPr><p:grpSpPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="0" cy="0"/><a:chOff x="0" y="0"/><a:chExt cx="0" cy="0"/></a:xfrm></p:grpSpPr>`
 ns:=` xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships" xmlns:p="http://schemas.openxmlformats.org/presentationml/2006/main"`
 entries["ppt/slideMasters/slideMaster1.xml"]=head+`<p:sldMaster`+ns+`><p:cSld><p:spTree>`+group+`</p:spTree></p:cSld><p:clrMap accent1="accent1" accent2="accent2" accent3="accent3" accent4="accent4" accent5="accent5" accent6="accent6" bg1="lt1" bg2="lt2" folHlink="folHlink" hlink="hlink" tx1="dk1" tx2="dk2"/><p:sldLayoutIdLst><p:sldLayoutId id="2147483649" r:id="rId1"/></p:sldLayoutIdLst><p:txStyles><p:titleStyle/><p:bodyStyle/><p:otherStyle/></p:txStyles></p:sldMaster>`
 entries["ppt/slideMasters/_rels/slideMaster1.xml.rels"]=head+relHead+rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml")+`</Relationships>`
 entries["ppt/slideLayouts/slideLayout1.xml"]=head+`<p:sldLayout`+ns+` type="blank" preserve="1"><p:cSld name="Blank"><p:spTree>`+group+`</p:spTree></p:cSld><p:clrMapOvr><a:masterClrMapping/></p:clrMapOvr></p:sldLayout>`
 entries["ppt/slideLayouts/_rels/slideLayout1.xml.rels"]=head+relHead+rel("rId1","slideMaster","../slideMasters/slideMaster1.xml")+`</Relationships>`
 bodies:=[][]string{{"Synthetic QA briefing"},{facts[0]},{facts[1],facts[2]},{releaseStatus}}
 for i:=1;i<=4;i++ { entries[fmt.Sprintf("ppt/slides/slide%d.xml",i)]=slide(titles[i-1],bodies[i-1]); entries[fmt.Sprintf("ppt/slides/_rels/slide%d.xml.rels",i)]=head+relHead+rel("rId1","slideLayout","../slideLayouts/slideLayout1.xml")+`</Relationships>` }
 f,err:=os.Create(path); must(err); z:=zip.NewWriter(f)
 for name,data:=range entries { w,err:=z.Create(name); must(err); _,err=io.WriteString(w,data); must(err) }; must(z.Close()); must(f.Close())
}
func update(path string) {
 z,err:=zip.OpenReader(path); must(err)
 var buf bytes.Buffer; out:=zip.NewWriter(&buf); found:=false
 for _,f:=range z.File {
  if f.Name!="ppt/slides/slide4.xml" { must(out.Copy(f)); continue }
  r,err:=f.Open(); must(err); data,err:=io.ReadAll(r); must(err); must(r.Close())
  if !bytes.Contains(data,[]byte(oldStatus)) { panic("slide 4 old status missing") }
  data=bytes.Replace(data,[]byte(oldStatus),[]byte(releaseStatus),1); found=true
  h:=f.FileHeader; w,err:=out.CreateHeader(&h); must(err); _,err=w.Write(data); must(err)
 }
 must(out.Close()); must(z.Close()); if !found { panic("slide 4 missing") }; must(os.WriteFile(path,buf.Bytes(),0644))
}
func verify(path string) {
 z,err:=zip.OpenReader(path); must(err); defer z.Close()
 entries:=map[string][]byte{}; count:=0
 for _,f:=range z.File { r,err:=f.Open(); must(err); data,err:=io.ReadAll(r); must(err); must(r.Close()); entries[f.Name]=data
  if strings.HasSuffix(f.Name,".xml")||strings.HasSuffix(f.Name,".rels") { d:=xml.NewDecoder(bytes.NewReader(data)); for { _,err=d.Token(); if err==io.EOF { break }; must(err) } }
  if strings.HasPrefix(f.Name,"ppt/slides/slide")&&strings.HasSuffix(f.Name,".xml") { count++ }
 }
 if count!=4 { panic("expected four slides") }
 type override struct { Part string `xml:"PartName,attr"`; MIME string `xml:"ContentType,attr"` }; var ct struct { Overrides []override `xml:"Override"` }; must(xml.Unmarshal(entries["[Content_Types].xml"],&ct)); ok:=false
 for _,v:=range ct.Overrides { if v.Part=="/ppt/presentation.xml"&&v.MIME==mime { ok=true } }; if !ok { panic("incorrect presentation MIME") }
 all:=""
 for i,t:=range titles { data:=entries[fmt.Sprintf("ppt/slides/slide%d.xml",i+1)]; d:=xml.NewDecoder(bytes.NewReader(data)); texts:=[]string{}; shapes:=0
  for { tok,err:=d.Token(); if err==io.EOF { break }; must(err); if v,ok:=tok.(xml.StartElement); ok { if v.Name.Local=="sp" { shapes++ }; if v.Name.Local=="t" { var text string; must(d.DecodeElement(&text,&v)); texts=append(texts,text) } } }
  if shapes<3||len(texts)==0||texts[0]!=t { panic("missing editable title or shapes") }; joined:=strings.Join(texts,"\n"); if !strings.Contains(joined,disclosure) { panic("missing disclosure") }; if i==3&&!strings.Contains(joined,releaseStatus) { panic("missing status") }; all+=joined+"\n"
 }
 for _,fact:=range facts { if !strings.Contains(all,fact) { panic("missing synthetic fact") } }
 fmt.Printf("Verified saved file: %s\nSlide count: %d\nPresentation MIME: %s\nAll required titles, synthetic facts, disclosure, and editable text shapes verified.\n",path,count,mime)
}
func main() { if len(os.Args)!=2 { panic("usage: create_live_deck target.pptx") }; path:=os.Args[1]; _,err:=os.Stat(path); if os.IsNotExist(err) { create(path) } else { must(err); update(path) }; verify(path) }
