package core

import (
	"context"
	"testing"
)

// TestAudit_PlanModeInterceptorsCrossAssemblies models two simultaneous
// AssembleAgent calls: each call registers its own controller in the process
// global interceptor list. A write originating from the second (inactive)
// assembly is still blocked by the first assembly's active plan mode.
func TestAudit_PlanModeInterceptorsCrossAssemblies(t *testing.T) {
	ClearToolInterceptors()
	t.Cleanup(ClearToolInterceptors)

	ctx := context.Background()
	activeSession := NewDefaultPlanModeController()
	inactiveSession := NewDefaultPlanModeController()
	if err := activeSession.Enter(ctx, "session A plan"); err != nil {
		t.Fatal(err)
	}

	// AssembleAgent registers one interceptor per assembly via
	// RegisterToolInterceptor (assemble.go:1383).
	RegisterToolInterceptor(NewPlanModeToolInterceptor(activeSession))
	RegisterToolInterceptor(NewPlanModeToolInterceptor(inactiveSession))

	// PreToolCallEvent has no session/controller field and invokes the global
	// list, so this event for session B is blocked by session A.
	event := &PreToolCallEvent{ToolName: "write", ToolCallID: "session-b-write"}
	if !event.IsCancelled() {
		t.Fatal("expected process-global plan mode from session A to block session B")
	}
}
